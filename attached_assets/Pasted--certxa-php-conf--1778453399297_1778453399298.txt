# =============================================================================
#  certxa-php.conf  –  Certxa PHP Site Configuration
#  Points certxa.com to the PHP site running on port 8422
# =============================================================================

# ── HTTP → HTTPS redirect ─────────────────────────────────────────────────────
server {
    listen 80;
    listen [::]:80;
    server_name certxa.com www.certxa.com;
    return 301 https://$host$request_uri;
}

# ── Redirect www → non-www HTTPS ─────────────────────────────────────────────
server {
    listen 443 ssl http2;
    listen [::]:443 ssl http2;
    server_name www.certxa.com;

    ssl_certificate     /etc/letsencrypt/live/certxa.com-0001/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/certxa.com-0001/privkey.pem;
    ssl_trusted_certificate /etc/letsencrypt/live/certxa.com-0001/chain.pem;
    include /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;

    # OCSP stapling — resolver needed to look up OCSP endpoint
    ssl_stapling on;
    ssl_stapling_verify on;
    resolver 8.8.8.8 8.8.4.4 valid=300s;
    resolver_timeout 5s;

    return 301 https://certxa.com$request_uri;
}


# ── Main HTTPS server ─────────────────────────────────────────────────────────
server {
    listen 443 ssl http2;
    listen [::]:443 ssl http2;

    # Root domain only — www is permanently redirected above
    server_name certxa.com;

    client_max_body_size 50M;

    # ── SSL ──────────────────────────────────────────────────────────────────
    ssl_certificate     /etc/letsencrypt/live/certxa.com-0001/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/certxa.com-0001/privkey.pem;
    ssl_trusted_certificate /etc/letsencrypt/live/certxa.com-0001/chain.pem;

    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-GCM-SHA256:ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384;
    ssl_prefer_server_ciphers off;
    ssl_session_cache shared:SSL:10m;
    ssl_session_timeout 1d;
    ssl_session_tickets off;

    # OCSP stapling — resolver needed to look up OCSP endpoint
    ssl_stapling on;
    ssl_stapling_verify on;
    resolver 8.8.8.8 8.8.4.4 valid=300s;
    resolver_timeout 5s;

    # ── Security headers ─────────────────────────────────────────────────────
    add_header Strict-Transport-Security "max-age=63072000; includeSubDomains; preload" always;
    add_header X-Frame-Options SAMEORIGIN always;
    add_header X-Content-Type-Options nosniff always;
    add_header Referrer-Policy strict-origin-when-cross-origin always;

    # ── PHP Site ─────────────────────────────────────────────────────────────
    # Proxy to PHP site running on port 8422
    location / {
        proxy_pass         http://127.0.0.1:8422;
        proxy_http_version 1.1;
        
        proxy_set_header   Host              $host;
        proxy_set_header   X-Real-IP         $remote_addr;
        proxy_set_header   X-Forwarded-For   $proxy_add_x_forwarded_for;
        proxy_set_header   X-Forwarded-Proto $scheme;
        proxy_set_header   X-Forwarded-Host  $host;

        proxy_read_timeout 30s;
        proxy_connect_timeout 10s;
        proxy_send_timeout 30s;
    }

    # ── Logs ─────────────────────────────────────────────────────────────────
    access_log /var/log/nginx/certxa_php_access.log;
    error_log  /var/log/nginx/certxa_php_error.log warn;
}
